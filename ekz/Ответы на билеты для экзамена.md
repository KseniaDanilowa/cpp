## Билет 1

### Теория 1: Передача параметров в функцию

**Передача по значению:**
- Создается копия переменной (значение копируется в новый участок памяти)
- Изменения параметра внутри функции не влияют на оригинальную переменную
- Используется для простых типов (int, double, char) и когда не нужно изменять оригинал

**Пример:**
```cpp
void increment(int x) {
    x++;  // Изменяется только копия
}
int main() {
    int a = 5;
    increment(a);
    cout << a;  // Выведет 5, а не 6
}
```

**Передача по ссылке:**
- Параметр является псевдонимом (алиасом) оригинальной переменной
- Изменения параметра напрямую влияют на оригинальную переменную
- Не создается копия, что эффективнее для больших объектов
- Ссылка должна быть инициализирована при создании

**Пример:**
```cpp
void increment(int& x) {
    x++;  // Изменяется оригинальная переменная
}
int main() {
    int a = 5;
    increment(a);
    cout << a;  // Выведет 6
}
```

**Передача по указателю:**
- Передается адрес переменной в памяти
- Можно передать nullptr (отсутствие значения)
- Требуется разыменование (*) для доступа к значению
- Указатель можно переназначить на другой объект

**Пример:**
```cpp
void increment(int* x) {
    if (x != nullptr) {  // Проверка на nullptr
        (*x)++;  // Разыменование указателя
    }
}
int main() {
    int a = 5;
    increment(&a);  // Передача адреса
    cout << a;  // Выведет 6
}
```

**Когда использовать const при передаче по ссылке:**
- Когда нужно передать большой объект без копирования, но не нужно его изменять
- Защищает от случайного изменения данных
- Позволяет передавать временные объекты

**Пример:**
```cpp
void print(const string& str) {  // const ссылка
    // str нельзя изменить, но копирования нет
    cout << str;
}
```

### Теория 2: Конструктор и деструктор

**Конструктор:**
- Специальный метод класса, который вызывается автоматически при создании объекта
- Имя совпадает с именем класса, не имеет типа возвращаемого значения
- Используется для инициализации полей объекта
- Может быть перегружен (несколько конструкторов с разными параметрами)

**Деструктор:**
- Специальный метод класса, который вызывается автоматически при уничтожении объекта
- Имя: ~ИмяКласса, не имеет параметров и типа возвращаемого значения
- Используется для освобождения ресурсов (память, файлы, сетевые соединения)
- В классе может быть только один деструктор

**Конструктор копирования:**
- Создает новый объект как копию существующего
- Вызывается при: передаче объекта в функцию по значению, возврате объекта из функции по значению, инициализации одного объекта другим
- Синтаксис: `ClassName(const ClassName& other)`

**Пример вызова конструктора копирования:**
```cpp
Student s1("Иван", 5);
Student s2 = s1;  // Вызывается конструктор копирования
Student s3(s1);   // Тоже вызывается конструктор копирования
```

### Практика: Класс Student

```cpp
#include <iostream>
#include <string>
using namespace std;

// Класс Student для хранения информации о студенте
class Student {
private:
    string name;    // Приватное поле: имя студента (инкапсуляция - защита данных)
    int grade;      // Приватное поле: оценка студента
    
public:
    // Конструктор с параметрами (инициализирует поля объекта при создании)
    // Список инициализации (после :) более эффективен, чем присваивание в теле
    Student(const string& studentName, int studentGrade) 
        : name(studentName), grade(studentGrade) {
        // Проверка корректности оценки (валидация данных)
        if (grade < 0 || grade > 5) {
            grade = 0;  // Установка значения по умолчанию при некорректном вводе
        }
    }
    
    // Конструктор по умолчанию (создает объект с пустым именем и нулевой оценкой)
    Student() : name(""), grade(0) {}
    
    // Геттер для имени (метод получения значения приватного поля)
    // const означает, что метод не изменяет объект
    string getName() const {
        return name;  // Возвращаем копию строки
    }
    
    // Геттер для оценки
    int getGrade() const {
        return grade;
    }
    
    // Сеттер для имени (метод установки значения приватного поля)
    // Принимает const ссылку для избежания копирования строки
    void setName(const string& studentName) {
        if (!studentName.empty()) {  // Проверка на пустую строку
            name = studentName;
        }
    }
    
    // Сеттер для оценки с валидацией
    void setGrade(int studentGrade) {
        if (studentGrade >= 0 && studentGrade <= 5) {  // Проверка диапазона
            grade = studentGrade;
        } else {
            cout << "Ошибка: оценка должна быть от 0 до 5" << endl;
        }
    }
    
    // Метод для вывода информации о студенте
    void print() const {  // const метод - не изменяет объект
        cout << "Студент: " << name << ", Оценка: " << grade << endl;
    }
};

int main() {
    // Создание объектов класса Student (вызывается конструктор)
    Student student1("Иван Петров", 5);  // Создание с параметрами
    Student student2("Мария Сидорова", 4);
    Student student3;  // Создание с конструктором по умолчанию
    
    // Демонстрация работы методов получения данных (геттеры)
    cout << "=== Использование геттеров ===" << endl;
    cout << "Имя первого студента: " << student1.getName() << endl;
    cout << "Оценка первого студента: " << student1.getGrade() << endl;
    
    // Демонстрация работы методов изменения данных (сеттеры)
    cout << "\n=== Использование сеттеров ===" << endl;
    student3.setName("Петр Иванов");
    student3.setGrade(3);
    student3.print();  // Вывод информации о третьем студенте
    
    // Изменение данных первого студента
    student1.setGrade(4);  // Изменение оценки
    cout << "\nПосле изменения оценки:" << endl;
    student1.print();
    
    // Демонстрация работы метода print() для всех студентов
    cout << "\n=== Информация о всех студентах ===" << endl;
    student1.print();
    student2.print();
    student3.print();
    
    // Демонстрация валидации (проверки корректности данных)
    cout << "\n=== Проверка валидации ===" << endl;
    student1.setGrade(10);  // Некорректная оценка (должна быть от 0 до 5)
    cout << "Оценка после некорректного ввода: " << student1.getGrade() << endl;
    // Оценка не изменилась, так как валидация отклонила значение
    
    return 0;  // При выходе из main() автоматически вызываются деструкторы для всех объектов
}
```

## Билет 2

### Теория 1: Указатели и ссылки

**Указатель:**
- Переменная, которая хранит адрес другой переменной в памяти
- Может быть переназначен на другой объект
- Может иметь значение nullptr (не указывать ни на что)
- Требует операцию разыменования (*) для доступа к значению
- Занимает память (обычно 4 или 8 байт в зависимости от системы)

**Ссылка:**
- Псевдоним (алиас) для существующей переменной
- Должна быть инициализирована при создании (нельзя создать ссылку без инициализации)
- Нельзя переназначить на другой объект после инициализации
- Не может быть nullptr
- Не требует разыменования для доступа к значению
- Не занимает отдельную память (компилятор работает с оригинальной переменной)

**Сходства:**
- Оба позволяют работать с объектом косвенно (не напрямую)
- Оба могут использоваться для передачи параметров в функции
- Оба позволяют избежать копирования больших объектов

**Ключевые отличия:**
1. Инициализация: ссылка обязательна, указатель может быть nullptr
2. Переназначение: ссылку нельзя переназначить, указатель можно
3. Синтаксис: ссылка используется как обычная переменная, указатель требует *
4. Память: ссылка не занимает память, указатель занимает

**Можно ли создать ссылку без объекта?**
НЕТ! Ссылка должна быть инициализирована при создании. Попытка создать ссылку без инициализации приведет к ошибке компиляции.

```cpp
int& ref;  // ОШИБКА КОМПИЛЯЦИИ!
int* ptr = nullptr;  // OK - указатель может быть nullptr
```

### Теория 2: Перегрузка функций и операторов

**Перегрузка функций:**
- Возможность создать несколько функций с одинаковым именем, но разными параметрами
- Компилятор выбирает нужную функцию на основе типов и количества аргументов
- Параметры должны отличаться по типу или количеству

**Пример перегрузки функции:**
```cpp
int add(int a, int b) {
    return a + b;
}

double add(double a, double b) {
    return a + b;
}

string add(const string& a, const string& b) {
    return a + b;  // Конкатенация строк
}
```

**Перегрузка операторов:**
- Позволяет определить поведение операторов (+, -, *, == и т.д.) для пользовательских типов
- Может быть реализована как метод класса или как дружественная функция
- Операторы ввода/вывода (<<, >>) обычно перегружаются как дружественные функции

**Пример перегрузки оператора +:**
```cpp
class Point {
private:
    int x, y;
public:
    Point(int x, int y) : x(x), y(y) {}
    
    // Перегрузка оператора + как метод класса
    Point operator+(const Point& other) const {
        return Point(x + other.x, y + other.y);
    }
    
    // Или как дружественная функция
    friend Point operator+(const Point& p1, const Point& p2);
};

Point operator+(const Point& p1, const Point& p2) {
    return Point(p1.x + p2.x, p1.y + p2.y);
}
```

### Практика: Класс Array

```cpp
#include <iostream>
#include <stdexcept>  // Для std::out_of_range
using namespace std;

// Класс Array для работы с динамическим массивом целых чисел
class Array {
private:
    int* data;        // Указатель на динамический массив (выделенная память в heap)
    int size;         // Размер массива (количество элементов)
    
public:
    // Конструктор с параметром размера (выделяет память для массива)
    // explicit предотвращает неявное преобразование int в Array
    explicit Array(int arraySize) : size(arraySize) {
        if (arraySize <= 0) {
            throw invalid_argument("Размер массива должен быть положительным");
        }
        // Выделение динамической памяти для массива (new выделяет память в heap)
        data = new int[size];  // new[] выделяет память для массива
        
        // Инициализация всех элементов нулями
        for (int i = 0; i < size; i++) {
            data[i] = 0;
        }
    }
    
    // Конструктор копирования (создает копию массива - глубокое копирование)
    Array(const Array& other) : size(other.size) {
        data = new int[size];  // Выделяем новую память
        // Копируем все элементы из другого массива
        for (int i = 0; i < size; i++) {
            data[i] = other.data[i];
        }
    }
    
    // Оператор присваивания (правило трех - нужен при наличии деструктора)
    Array& operator=(const Array& other) {
        if (this != &other) {  // Проверка на самоприсваивание (a = a)
            delete[] data;      // Освобождаем старую память
            size = other.size;
            data = new int[size];  // Выделяем новую память
            // Копируем элементы
            for (int i = 0; i < size; i++) {
                data[i] = other.data[i];
            }
        }
        return *this;  // Возвращаем ссылку на текущий объект
    }
    
    // Деструктор (освобождает выделенную память)
    // Вызывается автоматически при уничтожении объекта
    ~Array() {
        delete[] data;  // delete[] освобождает память, выделенную new[]
        data = nullptr;  // Хорошая практика - обнулить указатель после удаления
    }
    
    // Метод получения размера массива
    int getSize() const {
        return size;
    }
    
    // Метод получения значения по индексу с проверкой границ
    // const метод - не изменяет объект
    int getValue(int index) const {
        if (index < 0 || index >= size) {  // Проверка выхода за границы
            throw out_of_range("Индекс выходит за границы массива");
        }
        return data[index];  // Возвращаем значение элемента
    }
    
    // Метод изменения значения по индексу с проверкой границ
    void setValue(int index, int value) {
        if (index < 0 || index >= size) {  // Проверка выхода за границы
            throw out_of_range("Индекс выходит за границы массива");
        }
        data[index] = value;  // Устанавливаем новое значение
    }
    
    // Перегрузка оператора [] для удобного доступа к элементам
    // Не-const версия (позволяет изменять элементы)
    int& operator[](int index) {
        if (index < 0 || index >= size) {
            throw out_of_range("Индекс выходит за границы массива");
        }
        return data[index];  // Возвращаем ссылку на элемент (можно изменять)
    }
    
    // Const версия оператора [] (для const объектов)
    const int& operator[](int index) const {
        if (index < 0 || index >= size) {
            throw out_of_range("Индекс выходит за границы массива");
        }
        return data[index];  // Возвращаем const ссылку (нельзя изменять)
    }
    
    // Метод для вывода массива (для демонстрации)
    void print() const {
        cout << "Массив [";
        for (int i = 0; i < size; i++) {
            cout << data[i];
            if (i < size - 1) cout << ", ";
        }
        cout << "]" << endl;
    }
};

int main() {
    try {
        // Создание массива размером 5 (вызывается конструктор)
        Array arr(5);
        
        cout << "Размер массива: " << arr.getSize() << endl;
        
        // Установка значений через метод setValue
        cout << "\n=== Установка значений ===" << endl;
        arr.setValue(0, 10);
        arr.setValue(1, 20);
        arr.setValue(2, 30);
        arr.setValue(3, 40);
        arr.setValue(4, 50);
        arr.print();
        
        // Получение значений через метод getValue
        cout << "\n=== Получение значений ===" << endl;
        cout << "Элемент [0]: " << arr.getValue(0) << endl;
        cout << "Элемент [2]: " << arr.getValue(2) << endl;
        
        // Использование оператора [] для доступа к элементам
        cout << "\n=== Использование оператора [] ===" << endl;
        arr[1] = 25;  // Изменение через оператор []
        cout << "Элемент [1] после изменения: " << arr[1] << endl;
        arr.print();
        
        // Демонстрация конструктора копирования
        cout << "\n=== Конструктор копирования ===" << endl;
        Array arr2 = arr;  // Создается копия массива
        arr2[0] = 100;     // Изменение в копии не влияет на оригинал
        cout << "Оригинальный массив: ";
        arr.print();
        cout << "Скопированный массив: ";
        arr2.print();
        
        // Демонстрация проверки границ (обработка исключения)
        cout << "\n=== Проверка границ ===" << endl;
        try {
            arr.setValue(10, 100);  // Попытка доступа за границами
        } catch (const out_of_range& e) {
            cout << "Ошибка: " << e.what() << endl;
        }
        
    } catch (const exception& e) {
        cout << "Ошибка: " << e.what() << endl;
    }
    
    // При выходе из main() автоматически вызывается деструктор для arr
    // Деструктор освобождает выделенную память (нет утечки памяти)
    
    return 0;
}
```

## Билет 3

### Теория 1: Три основных принципа ООП

**1. Инкапсуляция:**
- Объединение данных (полей) и методов (функций) в одной единице - классе
- Сокрытие внутренней реализации от внешнего кода
- Контроль доступа через модификаторы доступа (private, protected, public)
- Защита данных от некорректного изменения через публичные методы (геттеры/сеттеры)

**2. Наследование:**
- Механизм создания нового класса на основе существующего (базового)
- Производный класс наследует поля и методы базового класса
- Позволяет переиспользовать код и расширять функциональность
- Поддерживает иерархию классов (отношение "is-a" - "является")

**3. Полиморфизм:**
- Возможность объектов разных классов обрабатываться как объекты общего базового класса
- Один интерфейс - разные реализации
- Достигается через виртуальные функции
- Позволяет вызывать правильный метод в зависимости от типа объекта во время выполнения

### Теория 2: Виртуальные функции

**Виртуальные функции:**
- Функции, объявленные с ключевым словом `virtual` в базовом классе
- Позволяют производным классам переопределять поведение функции
- Выбор конкретной реализации происходит во время выполнения (динамическое связывание)

**Зачем нужен virtual:**
- Без virtual: вызывается метод базового класса (статическое связывание)
- С virtual: вызывается метод производного класса (динамическое связывание)
- Обеспечивает правильное полиморфное поведение

**Чисто виртуальная функция:**
- Функция, объявленная как `virtual ... = 0;`
- Не имеет реализации в базовом классе
- Обязана быть реализована в производных классах
- Делает класс абстрактным (нельзя создать объект такого класса)

**Абстрактный класс:**
- Класс, содержащий хотя бы одну чисто виртуальную функцию
- Нельзя создать объект абстрактного класса
- Используется как интерфейс или базовый класс для иерархии

### Практика: Иерархия классов Animal

```cpp
#include <iostream>
#include <string>
using namespace std;

// Базовый класс Animal (животное)
class Animal {
protected:
    // protected поля доступны в производных классах, но не извне
    string name;  // Имя животного
    
public:
    // Конструктор базового класса
    Animal(const string& animalName) : name(animalName) {}
    
    // Виртуальный метод (virtual позволяет переопределение в производных классах)
    // = 0 делает функцию чисто виртуальной (класс становится абстрактным)
    virtual void makeSound() const = 0;  // Чисто виртуальная функция
    
    // Виртуальный деструктор (обязателен для полиморфизма)
    // Гарантирует правильное уничтожение объектов производных классов
    virtual ~Animal() {
        cout << "Деструктор Animal для " << name << endl;
    }
    
    // Обычный метод (не виртуальный)
    string getName() const {
        return name;
    }
};

// Производный класс Dog (наследуется от Animal)
// public наследование означает, что public методы Animal остаются public
class Dog : public Animal {
public:
    // Конструктор производного класса
    // Вызывает конструктор базового класса через список инициализации
    Dog(const string& dogName) : Animal(dogName) {}
    
    // Переопределение виртуального метода (override явно указывает на переопределение)
    void makeSound() const override {
        cout << name << " гавкает: Гав-гав!" << endl;
    }
    
    // Деструктор производного класса
    ~Dog() {
        cout << "Деструктор Dog для " << name << endl;
    }
};

// Производный класс Cat (наследуется от Animal)
class Cat : public Animal {
public:
    // Конструктор производного класса
    Cat(const string& catName) : Animal(catName) {}
    
    // Переопределение виртуального метода (своя реализация для кота)
    void makeSound() const override {
        cout << name << " мяукает: Мяу-мяу!" << endl;
    }
    
    // Деструктор производного класса
    ~Cat() {
        cout << "Деструктор Cat для " << name << endl;
    }
};

int main() {
    // Создание массива указателей на базовый класс Animal
    // Указатели позволяют хранить объекты разных производных классов
    const int SIZE = 4;
    Animal* animals[SIZE];  // Массив указателей на Animal
    
    // Создание объектов производных классов
    // new выделяет память в heap (динамическая память)
    animals[0] = new Dog("Бобик");      // Создаем объект Dog
    animals[1] = new Cat("Мурка");      // Создаем объект Cat
    animals[2] = new Dog("Шарик");      // Еще один Dog
    animals[3] = new Cat("Васька");     // Еще один Cat
    
    cout << "=== Полиморфное поведение ===" << endl;
    // Полиморфизм: вызывается правильный метод в зависимости от типа объекта
    // Благодаря virtual, вызывается makeSound() производного класса, а не базового
    for (int i = 0; i < SIZE; i++) {
        animals[i]->makeSound();  // Вызов виртуального метода
        // Для Dog вызовется Dog::makeSound()
        // Для Cat вызовется Cat::makeSound()
    }
    
    cout << "\n=== Информация о животных ===" << endl;
    // Доступ к невиртуальному методу (одинаковый для всех)
    for (int i = 0; i < SIZE; i++) {
        cout << "Имя: " << animals[i]->getName() << endl;
    }
    
    // Освобождение памяти (обязательно для динамически выделенных объектов)
    cout << "\n=== Освобождение памяти ===" << endl;
    for (int i = 0; i < SIZE; i++) {
        delete animals[i];  // Вызывается деструктор производного класса, затем базового
        // Благодаря виртуальному деструктору вызывается правильный деструктор
    }
    
    return 0;
}
```

**Объяснение полиморфизма:**
- Без `virtual`: все вызовы `makeSound()` вызвали бы метод базового класса (если бы он был реализован)
- С `virtual`: каждый объект вызывает свой собственный метод
- Это называется динамическим связыванием (решение принимается во время выполнения программы)

### Практика: Класс BankAccount с исключениями

```cpp
#include <iostream>
#include <string>
#include <stdexcept>
using namespace std;

// Собственный класс исключения для недостаточных средств
// Наследуется от std::exception для совместимости со стандартной библиотекой
class InsufficientFundsException : public exception {
private:
    double requestedAmount;   // Запрашиваемая сумма
    double currentBalance;   // Текущий баланс
    string message;          // Сообщение об ошибке
    
public:
    // Конструктор исключения (сохраняет информацию об ошибке)
    InsufficientFundsException(double requested, double balance) 
        : requestedAmount(requested), currentBalance(balance) {
        message = "Недостаточно средств! Запрошено: " + to_string(requested) +
                  ", доступно: " + to_string(balance);
    }
    
    // Переопределение метода what() для возврата сообщения об ошибке
    const char* what() const noexcept override {
        return message.c_str();
    }
    
    // Геттеры для получения детальной информации
    double getRequestedAmount() const { return requestedAmount; }
    double getCurrentBalance() const { return currentBalance; }
};

// Класс BankAccount (банковский счет)
class BankAccount {
private:
    string accountNumber;  // Номер счета
    double balance;       // Баланс счета
    
public:
    // Конструктор с параметрами
    BankAccount(const string& accNumber, double initialBalance = 0.0)
        : accountNumber(accNumber), balance(initialBalance) {
        if (initialBalance < 0) {
            throw invalid_argument("Начальный баланс не может быть отрицательным");
        }
    }
    
    // Геттер для номера счета
    string getAccountNumber() const {
        return accountNumber;
    }
    
    // Геттер для баланса
    double getBalance() const {
        return balance;
    }
    
    // Метод для внесения денег на счет
    void deposit(double amount) {
        if (amount <= 0) {
            throw invalid_argument("Сумма внесения должна быть положительной");
        }
        balance += amount;  // Увеличение баланса
        cout << "Внесено: " << amount << ". Новый баланс: " << balance << endl;
    }
    
    // Метод для снятия денег со счета
    void withdraw(double amount) {
        if (amount <= 0) {
            throw invalid_argument("Сумма снятия должна быть положительной");
        }
        
        // Проверка достаточности средств
        if (amount > balance) {
            // Выбрасываем собственное исключение
            throw InsufficientFundsException(amount, balance);
        }
        
        balance -= amount;  // Уменьшение баланса
        cout << "Снято: " << amount << ". Новый баланс: " << balance << endl;
    }
    
    // Метод для вывода информации о счете
    void printInfo() const {
        cout << "Счет: " << accountNumber << ", Баланс: " << balance << endl;
    }
};

int main() {
    try {
        // Создание банковского счета с начальным балансом
        BankAccount account("1234567890", 1000.0);
        account.printInfo();
        
        // Внесение денег
        cout << "\n=== Внесение денег ===" << endl;
        account.deposit(500.0);
        account.printInfo();
        
        // Снятие денег (успешное)
        cout << "\n=== Снятие денег (успешное) ===" << endl;
        account.withdraw(300.0);
        account.printInfo();
        
        // Попытка снять больше, чем есть на счете (вызовет исключение)
        cout << "\n=== Попытка снять больше, чем есть ===" << endl;
        try {
            account.withdraw(2000.0);  // Запрашиваем больше, чем есть
        } catch (const InsufficientFundsException& e) {
            // Обработка собственного исключения
            cout << "Ошибка: " << e.what() << endl;
            cout << "Детали: запрошено " << e.getRequestedAmount() 
                 << ", доступно " << e.getCurrentBalance() << endl;
        }
        
        account.printInfo();  // Баланс не изменился (транзакция не выполнена)
        
        // Еще одна успешная операция
        cout << "\n=== Еще одно успешное снятие ===" << endl;
        account.withdraw(500.0);
        account.printInfo();
        
    } catch (const exception& e) {
        // Обработка любых других исключений
        cerr << "Общая ошибка: " << e.what() << endl;
        return 1;
    }
    
    return 0;
}
```

### Практика: Класс Counter

```cpp
#include <iostream>
using namespace std;

// Класс Counter для подсчета количества созданных объектов
class Counter {
private:
    // Статическая переменная-член (общая для всех объектов класса)
    // Счетчик количества созданных объектов
    static int count;  // Объявление статической переменной (определение будет вне класса)
    
    int id;  // Уникальный идентификатор объекта
    
public:
    // Конструктор (увеличивает счетчик при создании объекта)
    Counter() : id(count + 1) {
        count++;  // Увеличиваем общий счетчик
        cout << "Создан объект #" << id << " (всего объектов: " << count << ")" << endl;
    }
    
    // Деструктор (уменьшает счетчик при уничтожении объекта)
    ~Counter() {
        count--;  // Уменьшаем общий счетчик
        cout << "Уничтожен объект #" << id << " (осталось объектов: " << count << ")" << endl;
    }
    
    // Статический метод для получения текущего количества объектов
    // Статический метод не имеет доступа к нестатическим членам (не имеет this)
    static int getCount() {
        return count;  // Возвращаем значение статической переменной
    }
    
    // Метод для получения ID объекта
    int getId() const {
        return id;
    }
};

// Определение и инициализация статической переменной (обязательно вне класса)
int Counter::count = 0;  // Инициализация нулем (до создания первого объекта)

int main() {
    cout << "Начальное количество объектов: " << Counter::getCount() << endl;
    
    // Создание объектов в разных областях видимости
    {
        cout << "\n=== Создание объектов в блоке ===" << endl;
        Counter c1;  // Создается первый объект
        Counter c2;  // Создается второй объект
        Counter c3;  // Создается третий объект
        
        cout << "\nТекущее количество объектов: " << Counter::getCount() << endl;
        cout << "ID объектов: " << c1.getId() << ", " << c2.getId() << ", " << c3.getId() << endl;
        
        // В конце блока все объекты уничтожаются (вызываются деструкторы)
    }
    
    cout << "\nПосле выхода из блока количество объектов: " << Counter::getCount() << endl;
    
    // Создание еще одного объекта
    cout << "\n=== Создание объекта в main ===" << endl;
    Counter c4;
    cout << "Текущее количество объектов: " << Counter::getCount() << endl;
    
    // При выходе из main() вызывается деструктор для c4
    return 0;
}
```

### Теория 1: Дружественные функции и классы

**Дружественные функции/классы** имеют доступ к private и protected членам класса.

**Когда использовать:**
- Перегрузка операторов ввода/вывода (<<, >>)
- Функции, которым нужен доступ к приватным данным, но они не должны быть методами класса
- Редко: для тестирования или оптимизации

### Теория 2: Виды наследования

**public наследование:**
- public → public
- protected → protected
- private → недоступно

**protected наследование:**
- public → protected
- protected → protected
- private → недоступно

**private наследование:**
- public → private
- protected → private
- private → недоступно

**Отличие:** public наследование сохраняет уровни доступа, остальные понижают.

### Практика: Класс Point с перегрузкой операторов

```cpp
#include <iostream>
using namespace std;

// Класс Point для представления точки на плоскости
class Point {
private:
    double x, y;  // Координаты точки
    
public:
    // Конструктор с параметрами
    Point(double xCoord = 0, double yCoord = 0) : x(xCoord), y(yCoord) {}
    
    // Геттеры для координат
    double getX() const { return x; }
    double getY() const { return y; }
    
    // Дружественная функция для перегрузки оператора == (равенство)
    // friend позволяет функции обращаться к private членам класса
    friend bool operator==(const Point& p1, const Point& p2);
    
    // Дружественная функция для перегрузки оператора + (сложение как векторов)
    friend Point operator+(const Point& p1, const Point& p2);
    
    // Дружественная функция для перегрузки оператора << (вывод)
    friend ostream& operator<<(ostream& os, const Point& p);
};

// Реализация оператора == (проверка равенства двух точек)
// Две точки равны, если их координаты совпадают
bool operator==(const Point& p1, const Point& p2) {
    return (p1.x == p2.x) && (p1.y == p2.y);  // Доступ к private членам благодаря friend
}

// Реализация оператора + (сложение точек как векторов)
// Результат: новая точка с суммой координат
Point operator+(const Point& p1, const Point& p2) {
    return Point(p1.x + p2.x, p1.y + p2.y);  // Создание новой точки
}

// Реализация оператора << (вывод точки в поток)
// Возвращает ссылку на поток для цепочки операций (cout << p1 << p2)
ostream& operator<<(ostream& os, const Point& p) {
    os << "(" << p.x << ", " << p.y << ")";  // Формат вывода: (x, y)
    return os;  // Возвращаем поток для цепочки
}

int main() {
    // Создание точек
    Point p1(3.0, 4.0);
    Point p2(1.0, 2.0);
```

```cpp

#include <iomanip>  // Для форматирования вывода
using namespace std;

// Класс Matrix для работы с матрицей фиксированного размера 3x3
class Matrix {
private:
    static const int SIZE = 3;  // Размер матрицы (константа класса)
    int data[SIZE][SIZE];       // Двумерный массив для хранения элементов
    
public:
    // Конструктор по умолчанию (инициализирует нулями)
    Matrix() {
        for (int i = 0; i < SIZE; i++) {
            for (int j = 0; j < SIZE; j++) {
                data[i][j] = 0;
            }
        }
    }
    
    // Конструктор с массивом значений
    Matrix(int values[SIZE][SIZE]) {
        for (int i = 0; i < SIZE; i++) {
            for (int j = 0; j < SIZE; j++) {
                data[i][j] = values[i][j];
            }
        }
    }
    
    // Перегрузка оператора + (сложение двух матриц)
    // Возвращает новую матрицу (результат сложения)
    Matrix operator+(const Matrix& other) const {
        Matrix result;  // Создаем новую матрицу для результата
        for (int i = 0; i < SIZE; i++) {
            for (int j = 0; j < SIZE; j++) {
                // Сложение соответствующих элементов
                result.data[i][j] = data[i][j] + other.data[i][j];
            }
        }
        return result;  // Возвращаем результат
    }
    
    // Перегрузка оператора * (умножение матрицы на число)
    // Умножаем каждый элемент матрицы на число
    Matrix operator*(int scalar) const {
        Matrix result;
        for (int i = 0; i < SIZE; i++) {
            for (int j = 0; j < SIZE; j++) {
                result.data[i][j] = data[i][j] * scalar;
            }
        }
        return result;
    }
    
    // Перегрузка оператора << для вывода матрицы
    friend ostream& operator<<(ostream& os, const Matrix& m) {
        for (int i = 0; i < SIZE; i++) {
            for (int j = 0; j < SIZE; j++) {
                os << setw(4) << m.data[i][j] << " ";  // setw(4) - ширина поля
            }
            os << endl;
        }
        return os;
    }
    
    // Метод для установки значения элемента
    void setValue(int row, int col, int value) {
        if (row >= 0 && row < SIZE && col >= 0 && col < SIZE) {
            data[row][col] = value;
        }
    }
    
    // Метод для получения значения элемента (const метод)
    int getValue(int row, int col) const {
        if (row >= 0 && row < SIZE && col >= 0 && col < SIZE) {
            return data[row][col];
        }
        return 0;
    }
};

    int values1[3][3] = {{1, 2, 3}, {4, 5, 6}, {7, 8, 9}};
    Matrix m1(values1);
    
    // Создание второй матрицы
    int values2[3][3] = {{9, 8, 7}, {6, 5, 4}, {3, 2, 1}};
    Matrix m2(values2);
    
    cout << "Матрица m1:" << endl;
    cout << m1 << endl;
    
    cout << "Матрица m2:" << endl;
    cout << m2 << endl;
    
    // Сложение матриц
    cout << "m1 + m2:" << endl;
    Matrix sum = m1 + m2;  // Использование перегруженного оператора +
    cout << sum << endl;
    
    // Умножение матрицы на число
    cout << "m1 * 2:" << endl;
    Matrix multiplied = m1 * 2;  // Использование перегруженного оператора *
    cout << multiplied << endl;
    
    // Цепочка операций

    Matrix result = (m1 + m2) * 3;
    cout << result << endl;
    
    return 0;
```

### Теория 1: Агрегация vs Композиция

**Агрегация ("has-a", слабая связь):**

### Теория 2: Анонимные объекты

**Анонимные объекты** - временные объекты без имени, создаются для немедленного использования.

**Использование:** передача в функции, возврат из функций, инициализация.

### Практика: Класс Date

```cpp
#include <iostream>
#include <string>
#include <iomanip>
#include <sstream>
using namespace std;

// Класс Date для работы с датами
class Date {
private:
    int day;    // День
    int month;  // Месяц
    int year;   // Год
    
    // Вспомогательный метод для проверки високосного года
    bool isLeapYear(int y) const {
        // Год високосный, если делится на 4, но не на 100, или делится на 400
        return (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
    }
    
    // Вспомогательный метод для получения количества дней в месяце
    int daysInMonth(int m, int y) const {
        switch (m) {
            case 1: case 3: case 5: case 7: case 8: case 10: case 12:
                return 31;  // Месяцы с 31 днем
            case 4: case 6: case 9: case 11:
                return 30;  // Месяцы с 30 днями
            case 2:
                return isLeapYear(y) ? 29 : 28;  // Февраль (28 или 29 дней)
            default:
                return 0;
        }
    }
    
public:
    // Конструктор с параметрами
    Date(int d, int m, int y) : day(d), month(m), year(y) {}
    
    // Метод для проверки корректности даты
    bool isValid() const {
        // Проверка года (разумные границы)
        if (year < 1 || year > 9999) {
            return false;
        }
        
        // Проверка месяца
        if (month < 1 || month > 12) {
            return false;
        }
        
        // Проверка дня (с учетом количества дней в месяце и високосных лет)
        if (day < 1 || day > daysInMonth(month, year)) {
            return false;
        }
        
        return true;  // Все проверки пройдены
    }
    
    // Метод для преобразования даты в строку формата "ДД.ММ.ГГГГ"
    string toString() const {
        if (!isValid()) {
            return "Некорректная дата";
        }
        
        // Использование stringstream для форматирования
        stringstream ss;
        ss << setfill('0') << setw(2) << day << "."    // День с ведущим нулем
           << setw(2) << month << "."                  // Месяц с ведущим нулем
           << setw(4) << year;                        // Год (4 цифры)
        
        return ss.str();  // Преобразование stream в string
    }
    
    // Геттеры
    int getDay() const { return day; }
    int getMonth() const { return month; }
    int getYear() const { return year; }
};

int main() {
    // Тестирование различных дат
    cout << "=== Проверка корректности дат ===" << endl;
    
    Date date1(15, 5, 2023);  // Корректная дата
    cout << "Дата 1: " << date1.toString() 
         << " - " << (date1.isValid() ? "корректна" : "некорректна") << endl;
    
    Date date2(29, 2, 2024);  // 29 февраля в високосном году
    cout << "Дата 2: " << date2.toString() 
         << " - " << (date2.isValid() ? "корректна" : "некорректна") << endl;
    
    Date date3(29, 2, 2023);  // 29 февраля в невисокосном году (некорректно)
    cout << "Дата 3: " << date3.toString() 
         << " - " << (date3.isValid() ? "корректна" : "некорректна") << endl;
```

### Практика: Класс Fraction

```cpp
#include <iostream>
#include <algorithm>  // Для __gcd (или можно написать свою функцию)
using namespace std;

// Класс Fraction (дробь)
class Fraction {
private:
    int numerator;    // Числитель
    int denominator;  // Знаменатель
    
    // Вспомогательный метод для вычисления НОД (наибольший общий делитель)
    int gcd(int a, int b) const {
        // Алгоритм Евклида для нахождения НОД
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
    
    // Метод для упрощения дроби (сокращения)
    void simplify() {
        if (denominator == 0) {
            throw invalid_argument("Знаменатель не может быть нулем");
        }
        
        // Обработка отрицательных чисел
        if (denominator < 0) {
            numerator = -numerator;
            denominator = -denominator;
        }
        
        // Находим НОД и делим на него
        int commonDivisor = gcd(abs(numerator), denominator);
        numerator /= commonDivisor;
        denominator /= commonDivisor;
    }
    
public:
    // Конструктор с упрощением дроби
    Fraction(int num, int den) : numerator(num), denominator(den) {
        simplify();  // Упрощаем дробь при создании (например, 2/4 -> 1/2)
    }
    
    // Конструктор по умолчанию
    Fraction() : numerator(0), denominator(1) {}
    
    // Геттеры
    int getNumerator() const { return numerator; }
    int getDenominator() const { return denominator; }
    
    // Перегрузка оператора + (сложение дробей)
    Fraction operator+(const Fraction& other) const {
        // Формула: a/b + c/d = (a*d + c*b) / (b*d)
        int newNum = numerator * other.denominator + other.numerator * denominator;
        int newDen = denominator * other.denominator;
        return Fraction(newNum, newDen);  // Создается новая дробь (автоматически упрощается)
    }
    
    // Дружественная функция для перегрузки оператора << (вывод)
    friend ostream& operator<<(ostream& os, const Fraction& f) {
        os << f.numerator << "/" << f.denominator;  // Формат: числитель/знаменатель
        return os;  // Возвращаем поток для цепочки операций
    }
};

int main() {
    // Создание дробей (автоматически упрощаются)
    Fraction f1(2, 4);   // 2/4 упростится до 1/2
    Fraction f2(3, 6);   // 3/6 упростится до 1/2
    Fraction f3(1, 3);   // 1/3
    
    cout << "=== Создание дробей ===" << endl;
    cout << "f1 = " << f1 << endl;  // Использование перегруженного оператора <<
    cout << "f2 = " << f2 << endl;
    cout << "f3 = " << f3 << endl;
    
    // Сложение дробей
    cout << "\n=== Сложение дробей ===" << endl;
    Fraction sum1 = f1 + f2;  // 1/2 + 1/2 = 2/2 = 1/1
    cout << f1 << " + " << f2 << " = " << sum1 << endl;
    
    Fraction sum2 = f1 + f3;  // 1/2 + 1/3 = 5/6
    cout << f1 << " + " << f3 << " = " << sum2 << endl;
    
    // Цепочка операций
    Fraction sum3 = f1 + f2 + f3;
    cout << f1 << " + " << f2 << " + " << f3 << " = " << sum3 << endl;
    
    return 0;
}
```

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

// Шаблонная функция для подсчета вхождений значения в вектор
// template<typename T> - T может быть любым типом
template<typename T>
int countOccurrences(const vector<T>& vec, const T& value) {
    int count = 0;  // Счетчик вхождений
    
    // Перебор всех элементов вектора
    for (const T& element : vec) {
        if (element == value) {  // Сравнение элементов (требует оператор == для типа T)
            count++;  // Увеличиваем счетчик при совпадении
        }
    }
    
    return count;  // Возвращаем количество вхождений
}

    // Тест 2: Вектор строк
    cout << "\n=== Тест с string ===" << endl;

    // Тест 3: Элемент не найден

    int notFound = countOccurrences(intVec, 99);

    // Тест 4: Пустой вектор
    cout << "\n=== Тест: пустой вектор ===" << endl;
```

```cpp
#include <string>
```

```cpp
class Dictionary {
private:
    map<string, string> translations;  // map: ключ (англ.) -> значение (рус.)
    
public:
    // Метод для добавления пары слов
    void addTranslation(const string& english, const string& russian) {
        translations[english] = russian;  // Добавление или обновление перевода
        cout << "Добавлен перевод: " << english << " -> " << russian << endl;
    }
    
    // Метод для поиска перевода
    string findTranslation(const string& english) const {
        // Поиск в map по ключу
        auto it = translations.find(english);  // find возвращает итератор
        
        if (it != translations.end()) {  // Если найдено
            return it->second;  // Возвращаем значение (русский перевод)
        } else {
            return "Перевод не найден";
        }
    }
    
    // Метод для вывода всего словаря
    void printAll() const {
        if (translations.empty()) {
            cout << "Словарь пуст" << endl;
            return;
        }
        
        cout << "\n=== Весь словарь ===" << endl;
        // Перебор всех элементов map
        for (const auto& pair : translations) {
            // pair.first - ключ (английское слово)
            // pair.second - значение (русский перевод)
            cout << pair.first << " -> " << pair.second << endl;
        }
    }
    
    // Метод для получения количества слов
    size_t getWordCount() const {
        return translations.size();
    }
};

int main() {
    Dictionary dict;  // Создание словаря
    
    // Добавление переводов
```

### Практика: Класс String (правило трех)

```cpp
#include <iostream>
#include <cstring>  // Для strlen, strcpy
using namespace std;

// Класс String (упрощенная версия std::string)
// Реализует правило трех: деструктор, конструктор копирования, оператор присваивания
class String {
private:
    char* data;      // Указатель на динамический массив символов
    size_t length;   // Длина строки
    
public:
    // Конструктор (создает строку из C-строки)
    String(const char* str = "") {
        if (str == nullptr) {
            str = "";  // Обработка nullptr
        }
        length = strlen(str);           // Вычисляем длину
        data = new char[length + 1];     // Выделяем память (+1 для '\0')
        strcpy(data, str);               // Копируем строку
    }
    
    // Конструктор копирования (глубокое копирование)
    // Создает независимую копию строки
    String(const String& other) : length(other.length) {
        data = new char[length + 1];     // Выделяем новую память
        strcpy(data, other.data);        // Копируем содержимое
    }
    
    // Оператор присваивания (глубокое копирование)
    String& operator=(const String& other) {
        if (this != &other) {  // Проверка на самоприсваивание (a = a)
            delete[] data;      // Освобождаем старую память
            
            length = other.length;
            data = new char[length + 1];  // Выделяем новую память
            strcpy(data, other.data);     // Копируем содержимое
        }
        return *this;  // Возвращаем ссылку на текущий объект
    }
    
    // Деструктор (освобождает динамическую память)
    ~String() {
        delete[] data;  // Освобождаем выделенную память
        data = nullptr; // Хорошая практика
    }
    
    // Метод для получения длины строки
    size_t getLength() const {
        return length;
    }
    
    // Метод для получения C-строки
    const char* c_str() const {
        return data;
    }
    
    // Перегрузка оператора << для вывода
    friend ostream& operator<<(ostream& os, const String& str) {
        os << str.data;
        return os;
    }
};

int main() {
    // Создание строк
    String str1("Hello");
    String str2("World");
    
    cout << "str1: " << str1 << " (длина: " << str1.getLength() << ")" << endl;
    cout << "str2: " << str2 << " (длина: " << str2.getLength() << ")" << endl;
    
    // Конструктор копирования (глубокое копирование)
    cout << "\n=== Конструктор копирования ===" << endl;
    String str3 = str1;  // Создается копия
    cout << "str3 (копия str1): " << str3 << endl;
    
    // Изменение оригинала не влияет на копию (глубокое копирование работает)
    // Но у нас нет метода для изменения, поэтому просто демонстрируем независимость
    
    // Оператор присваивания
    cout << "\n=== Оператор присваивания ===" << endl;
    String str4;
    str4 = str2;  // Присваивание
    cout << "str4 (после str4 = str2): " << str4 << endl;
    
    // Демонстрация независимости копий
    cout << "\n=== Демонстрация независимости ===" << endl;
    cout << "str1: " << str1 << endl;
    cout << "str3 (копия str1): " << str3 << endl;
    cout << "str2: " << str2 << endl;
    cout << "str4 (копия str2): " << str4 << endl;
    
    return 0;
    // При выходе из main() автоматически вызываются деструкторы для всех объектов
    // Каждый деструктор освобождает свою память (нет двойного освобождения)
}
```

```cpp
#include <iostream>
#include <stack>
#include <string>
using namespace std;

// Функция для проверки корректности расстановки скобок
bool checkBrackets(const string& expression) {
    stack<char> bracketStack;  // Стек для хранения открывающих скобок
    
    // Перебор всех символов строки
    for (char ch : expression) {
        // Если встретили открывающую скобку - кладем в стек
        if (ch == '(' || ch == '[' || ch == '{') {
            bracketStack.push(ch);
        }
        // Если встретили закрывающую скобку
        else if (ch == ')' || ch == ']' || ch == '}') {
            // Проверяем, не пуст ли стек
            if (bracketStack.empty()) {
                return false;  // Закрывающая скобка без открывающей
            }
            
            // Получаем последнюю открывающую скобку из стека
            char top = bracketStack.top();
            bracketStack.pop();  // Удаляем ее из стека
            
            // Проверяем соответствие скобок
            if ((ch == ')' && top != '(') ||
                (ch == ']' && top != '[') ||
                (ch == '}' && top != '{')) {
                return false;  // Несоответствие типов скобок
            }
        }
        // Игнорируем все остальные символы
    }
    
    // Если стек пуст - все скобки правильно закрыты
    return bracketStack.empty();
}

int main() {
    // Тестирование различных выражений
    string expressions[] = {
        "()",           // Корректно
        "()[]{}",       // Корректно
        "([{}])",       // Корректно
        "((()))",       // Корректно
        "([)]",         // Некорректно
        "((())",        // Некорректно (не хватает закрывающей)
        "())",          // Некорректно (лишняя закрывающая)
        "(a + b) * c",  // Корректно (с другими символами)
        "{[()]}",       // Корректно
        "{{{{"          // Некорректно (только открывающие)
    };
    
    cout << "=== Проверка расстановки скобок ===" << endl;
    for (const string& expr : expressions) {
        bool result = checkBrackets(expr);
        cout << "\"" << expr << "\" - " 
             << (result ? "корректно" : "некорректно") << endl;
    }
    
    return 0;
}
```

### Практика: Класс FileHandler (RAII)

```cpp
#include <iostream>
#include <fstream>
#include <string>
using namespace std;

// Класс FileHandler (обработчик файлов) с использованием RAII
// RAII (Resource Acquisition Is Initialization) - получение ресурса при инициализации
// Ресурс (файл) автоматически освобождается в деструкторе
class FileHandler {
private:
    fstream file;    // Поток для работы с файлом
    string filename; // Имя файла
    
public:
    // Конструктор открывает файл
    FileHandler(const string& name) : filename(name) {
        file.open(filename, ios::in | ios::out);  // Открываем для чтения и записи
        
        if (!file.is_open()) {
            // Если не удалось открыть, создаем новый файл
            file.open(filename, ios::out);
            file.close();
            file.open(filename, ios::in | ios::out);
        }
        
        if (file.is_open()) {
            cout << "Файл \"" << filename << "\" открыт" << endl;
        } else {
            cerr << "Ошибка: не удалось открыть файл \"" << filename << "\"" << endl;
        }
    }
    
    // Деструктор закрывает файл (автоматически вызывается при уничтожении объекта)
    ~FileHandler() {
        if (file.is_open()) {
            file.close();
            cout << "Файл \"" << filename << "\" закрыт" << endl;
        }
    }
    
    // Метод для чтения всего содержимого файла
    string readAll() {
        if (!file.is_open()) {
            return "";
        }
        
        // Перемещаем указатель в начало файла
        file.seekg(0, ios::beg);
        
        string content;
        string line;
        
        // Читаем файл построчно
        while (getline(file, line)) {
            content += line + "\n";
        }
        
        return content;
    }
    
    // Метод для записи в файл
    void write(const string& data) {
        if (file.is_open()) {
            file << data;
            file.flush();  // Принудительная запись в файл
        }
    }
    
    // Метод для проверки, открыт ли файл
    bool isOpen() const {
        return file.is_open();
    }
};

int main() {
    {
        // Создание FileHandler в блоке
        // Файл автоматически откроется в конструкторе
        FileHandler handler("test.txt");
        
        if (handler.isOpen()) {
            // Запись в файл
            handler.write("Первая строка\n");
            handler.write("Вторая строка\n");
            handler.write("Третья строка\n");
            
            // Чтение из файла
            cout << "\n=== Содержимое файла ===" << endl;
            string content = handler.readAll();
            cout << content;
        }
        
        // При выходе из блока автоматически вызывается деструктор
        // Деструктор закрывает файл (RAII в действии)
    }
    
    cout << "\nФайл автоматически закрыт при выходе из блока" << endl;
    
    return 0;
}
```

**explicit** - запрещает неявное преобразование типов через конструктор.

### Практика: Класс Complex

```cpp
#include <iostream>
#include <iomanip>
using namespace std;

// Класс Complex для работы с комплексными числами
// Комплексное число: a + bi, где a - действительная часть, b - мнимая часть, i - мнимая единица
class Complex {
private:
    double real;      // Действительная часть
    double imaginary; // Мнимая часть
    
public:
    // explicit конструктор запрещает неявное преобразование double в Complex
    explicit Complex(double r = 0.0, double i = 0.0) : real(r), imaginary(i) {}
    
    // Геттеры
    double getReal() const { return real; }
    double getImaginary() const { return imaginary; }
    
    // Перегрузка оператора + (сложение комплексных чисел)
    // (a + bi) + (c + di) = (a + c) + (b + d)i
    Complex operator+(const Complex& other) const {
        return Complex(real + other.real, imaginary + other.imaginary);
    }
    
    // Перегрузка оператора - (вычитание комплексных чисел)
    // (a + bi) - (c + di) = (a - c) + (b - d)i
    Complex operator-(const Complex& other) const {
        return Complex(real - other.real, imaginary - other.imaginary);
    }
    
    // Перегрузка оператора * (умножение комплексных чисел)
    // (a + bi) * (c + di) = (ac - bd) + (ad + bc)i
    Complex operator*(const Complex& other) const {
        double newReal = real * other.real - imaginary * other.imaginary;
        double newImaginary = real * other.imaginary + imaginary * other.real;
        return Complex(newReal, newImaginary);
    }
    
    // Дружественная функция для перегрузки оператора << (вывод)
    friend ostream& operator<<(ostream& os, const Complex& c) {
        os << fixed << setprecision(2);
        os << c.real;
        if (c.imaginary >= 0) {
            os << " + " << c.imaginary << "i";
        } else {
            os << " - " << -c.imaginary << "i";
        }
        return os;
    }
};

int main() {

    Complex c1(3.0, 4.0);   // 3 + 4i

    cout << "c1 = " << c1 << endl;
    cout << "c2 = " << c2 << endl;

    Complex sum = c1 + c2;
    cout << "\nc1 + c2 = " << sum << endl;
    
    // Вычитание комплексных чисел

    // Умножение комплексных чисел
```

### Теория 2: Делегирующий конструктор

**Делегирующий конструктор (C++11)** - конструктор, который вызывает другой конструктор того же класса.

**Помогает:** избежать дублирования кода инициализации.

### Практика: Класс Statistics

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
using namespace std;

// Класс Statistics для вычисления статистики по числам
class Statistics {
private:
    vector<double> numbers;  // Вектор для хранения чисел
    double sum;             // Сумма чисел (для быстрого вычисления среднего)
    
    // Приватный метод для пересчета суммы
    void recalculateSum() {
        sum = accumulate(numbers.begin(), numbers.end(), 0.0);
    }
    
public:
    // Конструктор, принимающий вектор (основной конструктор)
    Statistics(const vector<double>& nums) : numbers(nums) {
        recalculateSum();
    }
    
    // Делегирующий конструктор - принимает массив и размер
    // Вызывает основной конструктор с вектором
    Statistics(double* arr, size_t size) 
        : Statistics(vector<double>(arr, arr + size)) {
        // Тело пустое - вся инициализация делегирована основному конструктору
    }
    
    // Конструктор по умолчанию (пустая статистика)
    Statistics() : sum(0.0) {}
    
    // Метод для добавления числа
    void addNumber(double num) {
        numbers.push_back(num);
        sum += num;  // Обновляем сумму
    }
    
    // Метод для вычисления среднего значения
    double getMean() const {
        if (numbers.empty()) {
            return 0.0;
        }
        return sum / numbers.size();  // Среднее = сумма / количество
    }
    
    // Метод для получения минимального значения
    double getMin() const {
        if (numbers.empty()) {
            return 0.0;
        }
        // min_element возвращает итератор на минимальный элемент
        return *min_element(numbers.begin(), numbers.end());
    }
    
    // Метод для получения максимального значения
    double getMax() const {
        if (numbers.empty()) {
            return 0.0;
        }
        // max_element возвращает итератор на максимальный элемент
        return *max_element(numbers.begin(), numbers.end());
    }
    
    // Метод для получения количества чисел
    size_t getCount() const {
        return numbers.size();
    }
    
    // Метод для вывода статистики
    void print() const {
        cout << "Количество чисел: " << getCount() << endl;
        if (getCount() > 0) {
            cout << "Среднее: " << getMean() << endl;
            cout << "Минимум: " << getMin() << endl;
            cout << "Максимум: " << getMax() << endl;
        }
    }
};

int main() {
    // Использование конструктора с вектором
    cout << "=== Статистика из вектора ===" << endl;
    vector<double> nums1 = {10.0, 20.0, 30.0, 40.0, 50.0};
    Statistics stats1(nums1);
    stats1.print();
    
    // Использование делегирующего конструктора с массивом
    cout << "\n=== Статистика из массива ===" << endl;
    double arr[] = {5.5, 15.5, 25.5, 35.5};
    Statistics stats2(arr, 4);  // Делегирующий конструктор
    stats2.print();
    
    // Использование конструктора по умолчанию и добавление чисел
    cout << "\n=== Статистика с добавлением чисел ===" << endl;
    Statistics stats3;
    stats3.addNumber(1.0);
    stats3.addNumber(2.0);
    stats3.addNumber(3.0);
    stats3.addNumber(4.0);
    stats3.addNumber(5.0);
    stats3.print();
    
    return 0;
}
```

### Практика: Шаблонный класс Box

```cpp
#include <iostream>
#include <string>
using namespace std;

// Шаблонный класс Box для хранения одного элемента любого типа
template<typename T>
class Box {
private:
    T* item;        // Указатель на элемент (nullptr если пуст)
    bool isEmpty;   // Флаг пустоты
    
public:
    // Конструктор по умолчанию (создает пустой бокс)
    Box() : item(nullptr), isEmpty(true) {}
    
    // Деструктор
    ~Box() {
        if (item != nullptr) {
            delete item;  // Освобождаем память, если элемент был
        }
    }
    
    // Метод для помещения значения в бокс
    void put(const T& value) {
        if (item != nullptr) {
            delete item;  // Удаляем старое значение, если было
        }
        item = new T(value);  // Создаем копию значения
        isEmpty = false;
    }
    
    // Метод для получения значения из бокса
    T get() const {
        if (isEmpty || item == nullptr) {
            throw runtime_error("Бокс пуст");
        }
        return *item;  // Возвращаем копию значения
    }
    
    // Метод для проверки, пуст ли бокс
    bool checkEmpty() const {
        return isEmpty || item == nullptr;
    }
};

int main() {
    // Тест 1: Box для int
    cout << "=== Box для int ===" << endl;
    Box<int> intBox;
    cout << "Бокс пуст: " << (intBox.checkEmpty() ? "Да" : "Нет") << endl;
    
    intBox.put(42);
    cout << "После put(42), бокс пуст: " << (intBox.checkEmpty() ? "Да" : "Нет") << endl;
    cout << "Значение в боксе: " << intBox.get() << endl;
    
    // Тест 2: Box для string
    cout << "\n=== Box для string ===" << endl;
    Box<string> stringBox;
    stringBox.put("Hello, World!");
    cout << "Значение в боксе: " << stringBox.get() << endl;
    
    // Тест 3: Замена значения
    cout << "\n=== Замена значения ===" << endl;
    intBox.put(100);
    cout << "Новое значение в боксе: " << intBox.get() << endl;
    
    // Тест 4: Попытка получить из пустого бокса
    cout << "\n=== Попытка получить из пустого бокса ===" << endl;
    Box<int> emptyBox;
    try {
        int value = emptyBox.get();
        cout << "Значение: " << value << endl;
    } catch (const exception& e) {
        cout << "Ошибка: " << e.what() << endl;
    }
    
    return 0;
}
```

### Практика: Функция splitString

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

// Функция для разделения строки по разделителю
// Возвращает вектор подстрок
vector<string> splitString(const string& str, char delimiter) {
    vector<string> result;  // Вектор для хранения результата
    string current;          // Текущая подстрока
    
    // Перебор всех символов строки
    for (char ch : str) {
        if (ch == delimiter) {
            // Если встретили разделитель
            if (!current.empty()) {  // Если текущая подстрока не пуста
                result.push_back(current);  // Добавляем ее в результат
                current.clear();            // Очищаем для следующей подстроки
            }
        } else {
            // Если не разделитель - добавляем символ к текущей подстроке
            current += ch;
        }
    }
    
    // Добавляем последнюю подстроку (после последнего разделителя)
    if (!current.empty()) {
        result.push_back(current);
    }
    
    return result;  // Возвращаем вектор подстрок
}

int main() {
    // Тест 1: Разделение по запятой
    cout << "=== Тест 1: Разделение по запятой ===" << endl;
    string str1 = "apple,banana,cherry,date";
    vector<string> result1 = splitString(str1, ',');
    
    cout << "Исходная строка: \"" << str1 << "\"" << endl;
    cout << "Результат разделения:" << endl;
    for (size_t i = 0; i < result1.size(); i++) {
        cout << "  [" << i << "] \"" << result1[i] << "\"" << endl;
    }
    
    // Тест 2: Разделение по пробелу
    cout << "\n=== Тест 2: Разделение по пробелу ===" << endl;
    string str2 = "Hello World C++ Programming";

    for (size_t i = 0; i < result2.size(); i++) {
        cout << "  [" << i << "] \"" << result2[i] << "\"" << endl;
    }
    
    // Тест 3: Разделитель в начале и конце
    cout << "\n=== Тест 3: Разделитель в начале и конце ===" << endl;
    string str3 = ",first,second,third,";
    vector<string> result3 = splitString(str3, ',');

    return 0;
```

### Теория 1: Множественное наследование

**Множественное наследование** - класс наследуется от нескольких базовых классов.

**Проблема "ромбовидного" наследования:**
- Базовый класс наследуется через несколько путей
- Неоднозначность при обращении к членам базового класса
- Решение: виртуальное наследование (virtual inheritance)

### Теория 2: Анонимное пространство имен

**Анонимное пространство имен** `namespace { ... }` - элементы видны только в текущем файле.

**Заменяет:** статические глобальные переменные/функции (в современном C++).

### Практика: Интерфейс Drawable

```cpp
#include <iostream>
#include <vector>
using namespace std;

// Интерфейсный класс Drawable (рисуемый объект)
// Чисто виртуальная функция делает класс абстрактным
class Drawable {
public:
    // Чисто виртуальная функция (обязана быть реализована в производных классах)
    virtual void draw() const = 0;
    
    // Виртуальный деструктор (обязателен для полиморфизма)
    virtual ~Drawable() {}
};

// Класс Circle (круг), реализующий интерфейс Drawable
class Circle : public Drawable {
private:
    double radius;  // Радиус круга
    
public:
    Circle(double r) : radius(r) {}
    
    // Реализация чисто виртуальной функции draw()
    void draw() const override {
        cout << "Рисую круг радиусом " << radius << endl;
    }
};

// Класс Square (квадрат), реализующий интерфейс Drawable
class Square : public Drawable {
private:
    double side;  // Сторона квадрата
    
public:
    Square(double s) : side(s) {}
    
    // Реализация чисто виртуальной функции draw()
    void draw() const override {
        cout << "Рисую квадрат со стороной " << side << endl;
    }
};

int main() {
    // Создание вектора указателей на базовый класс Drawable
    vector<Drawable*> shapes;
    
    // Добавление объектов разных типов (полиморфизм)
    shapes.push_back(new Circle(5.0));    // Круг
    shapes.push_back(new Square(4.0));    // Квадрат
    shapes.push_back(new Circle(3.0));    // Еще один круг
    shapes.push_back(new Square(6.0));    // Еще один квадрат
    
    cout << "=== Рисование всех фигур ===" << endl;
    // Полиморфный вызов метода draw() для всех объектов
    // Каждый объект вызывает свою реализацию draw()
    for (Drawable* shape : shapes) {
        shape->draw();  // Виртуальный вызов - вызывается правильный метод
    }
    
    // Освобождение памяти
    cout << "\n=== Освобождение памяти ===" << endl;
    for (Drawable* shape : shapes) {
        delete shape;  // Вызывается виртуальный деструктор
    }
    
    return 0;
}
```
